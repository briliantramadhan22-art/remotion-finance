// Each scene fades and scales in/out quickly (Sequence gives the scene's own duration)
const SceneFrame: React.FC<{ children: React.ReactNode }> = ({ children }) => {
  const frame = useCurrentFrame();
  const { durationInFrames } = useVideoConfig();
  const opacity = interpolate(frame, [0, 6, durationInFrames - 6, durationInFrames], [0, 1, 1, 0], clamp);
  const scale = interpolate(frame, [0, 10], [0.96, 1], { ...clamp, easing: Easing.out(Easing.cubic) });
  return (
    <AbsoluteFill style={{ opacity }}>
      <AbsoluteFill style={{ transform: `scale(${scale})` }}>{children}</AbsoluteFill>
    </AbsoluteFill>
  );

// [seconds from scene start, caption line]

const Caption: React.FC<{ caps: Cap[] }> = ({ caps }) => {
  const frame = useCurrentFrame();
  const { fps } = useVideoConfig();
  let idx = 0;
  caps.forEach(([t], i) => {
    if (frame >= sec(t)) idx = i;
  });
  const pop = spring({ frame: frame - sec(caps[idx][0]), fps, config: { damping: 14, stiffness: 220 } });
  return (
    <div style={{ position: "absolute", left: 70, right: 70, bottom: 330, display: "flex", justifyContent: "center" }}>
      <div

          padding: "20px 38px",

          transform: `scale(${0.9 + 0.1 * Math.min(pop, 1.15)})`,

      >
        {caps[idx][1]}
      </div>
    </div>
  );
};

              boxShadow: `0 0 12px ${color}`,
              opacity: fade,
              transform: `rotate(${t * (random(`cr${i}`) * 24 - 12)}deg)`,

/* Scenes 3-5 - The rule: a neon donut fills one bucket per scene      */

const R = 250;
const SW = 56;
const CIRC = 2 * Math.PI * R;
const SEG_GAP = 10;
const STARTS = [0, 50, 80];

const Donut: React.FC<{ active: number; at: number }> = ({ active, at }) => {
  const frame = useCurrentFrame();
  const grow = useEnter(at, 34);
  const pulse = 0.75 + 0.25 * Math.sin(frame / 5);
  return (
    <svg width={600} height={600} viewBox="0 0 600 600" style={{ overflow: "visible" }}>
      <g transform="rotate(-90 300 300)">
        {SEGMENTS.map((seg, i) => {
          const len = (seg.pct / 100) * CIRC - SEG_GAP;
          const off = -(STARTS[i] / 100) * CIRC;
          const fill = i < active ? 1 : i === active ? grow : 0;
          return (
            <React.Fragment key={seg.key}>
              <circle
                cx={300}
                cy={300}
                r={R}
                fill="none"
                stroke={seg.color}
                strokeOpacity={0.16}
                strokeWidth={SW}
                strokeDasharray={`${len} ${CIRC}`}
                strokeDashoffset={off}
              />
              <circle
                cx={300}
                cy={300}
                r={R}
                fill="none"
                stroke={seg.color}
                strokeWidth={SW}
                strokeDasharray={`${Math.max(len * fill, 0)} ${CIRC}`}
                strokeDashoffset={off}
                style={{ filter: `drop-shadow(0 0 ${i === active ? 22 * pulse : 10}px ${seg.color})` }}
              />
            </React.Fragment>
          );
        })}
      </g>
    </svg>
  );
};

const Chip: React.FC<{ text: string; color: string; delay: number }> = ({ text, color, delay }) => {
  const p = useEnter(delay, 0, true);
  return (

        transform: `scale(${0.5 + 0.5 * p}) translateY(${(1 - one(p)) * 30}px)`,
      }}
    >
      {text}
    </div>
  );
};

const RuleScene: React.FC<{
  index: number;
  pctAt: number;
  chipsAt: number;
  gap: number;
  intro?: boolean;
  confettiAt?: number;
}> = ({ index, pctAt, chipsAt, gap, intro, confettiAt }) => {
  const seg = SEGMENTS[index];
  const frame = useCurrentFrame();
  const pct = Math.round(
    interpolate(frame, [pctAt, pctAt + 32], [0, seg.pct], { ...clamp, easing: Easing.out(Easing.cubic) })
  );
  const numOpacity = interpolate(frame, [pctAt, pctAt + 6], [0, 1], clamp);
  const introOpacity = intro ? interpolate(frame, [0, 8, pctAt - 8, pctAt], [0, 1, 1, 0], clamp) : 0;
  const label = one(useEnter(pctAt + 6, 0, true));

  return (
    <AbsoluteFill style={{ alignItems: "center", paddingTop: 230 }}>
      {confettiAt !== undefined && <Confetti from={confettiAt} to={confettiAt + 56} />}
      <div style={{ position: "relative", width: 600, height: 600 }}>
        <Donut active={index} at={pctAt} />
        <div
          style={{
            position: "absolute",
            inset: 0,
            display: "flex",
            flexDirection: "column",
            alignItems: "center",
            justifyContent: "center",
          }}
        >
          {intro && (
            <div style={{ position: "absolute", fontFamily: MONO, fontWeight: 700, fontSize: 78, opacity: introOpacity }}>
              50/30/20
            </div>
          )}
          <div
            style={{
              fontFamily: MONO,
              fontWeight: 700,
              fontSize: 150,
              lineHeight: 1,
              color: seg.color,
              opacity: numOpacity,
              textShadow: `0 0 36px ${seg.color}99`,
            }}
          >
            {pct}%
          </div>
          <div style={{ fontSize: 50, fontWeight: 700, opacity: label }}>{seg.label}</div>
        </div>
      </div>
      <div style={{ marginTop: 70, display: "flex", flexDirection: "column", gap: 24, alignItems: "center" }}>
        {seg.items.map((text, i) => (
          <Chip key={text} text={text} color={seg.color} delay={chipsAt + i * gap} />
        ))}
      </div>
    </AbsoluteFill>
  );
};
